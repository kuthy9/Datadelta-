# requirements.txt — datadelta core dependencies
#
# For most users, installing the package is enough:
#   pip install datadelta-cli
#
# This file exists for contributors who want the core dependencies
# without installing the package itself:
#   pip install -r requirements.txt
#
# It mirrors [project.dependencies] in pyproject.toml. Optional
# dependencies are listed at the bottom, commented out, together with
# the pyproject extra that installs them.

# ── CLI framework ─────────────────────────────────────────────────────────────
# Typer turns Python type annotations into CLI flags automatically
typer>=0.12

# ── Terminal output ───────────────────────────────────────────────────────────
# Rich provides colors, tables, and live progress in the terminal
rich>=13

# ── Core data processing ──────────────────────────────────────────────────────
pandas>=2.0
numpy>=1.24               # imported directly by the diff and metrics engines

# ── Embedded SQL engine ───────────────────────────────────────────────────────
# DuckDB reads CSV / JSON / Parquet directly with a SQL query — no import step
duckdb>=0.10

# ── Statistical tests ─────────────────────────────────────────────────────────
# KS test (distribution shift), chi-square (categorical drift)
scipy>=1.12

# ── LLM API (default provider) ────────────────────────────────────────────────
# Used by --story (with --llm claude, the default) and `datadelta init`
anthropic>=0.25

# ── Templating ────────────────────────────────────────────────────────────────
# Used by --export to render the self-contained HTML report
jinja2>=3.1

# ── Excel support ─────────────────────────────────────────────────────────────
# openpyxl is the engine pandas uses for .xlsx files
openpyxl>=3.1

# ── YAML parsing ─────────────────────────────────────────────────────────────
# Used to load metrics.yaml
pyyaml>=6.0

# ── SQL database connections ──────────────────────────────────────────────────
# SQLAlchemy provides a unified interface for all SQL databases
# The actual driver (psycopg2, pymysql, pyodbc) must also be installed
sqlalchemy>=2.0
